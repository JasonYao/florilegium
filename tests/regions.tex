\documentclass[10pt]{article}
\usepackage[letterpaper,left=15mm,right=45mm,top=15mm,bottom=15mm]{geometry}
\pagestyle{empty}
\def\florilegiumseed{24680}
\usepackage[grown,density=0.4,intensity=0.8]{florilegium}

\directlua{
  local growth = require("florilegium-grow")
  local base = { { 0, 0, 10, 10 } }
  local cases = {
    { {}, 100 },
    { { { 4, 4, 6, 6 } }, 96 },
    { { { 2, 2, 4, 5 }, { 6, 3, 8, 7 } }, 86 },
  }
  for _, case in ipairs(cases) do
    local regions = growth.subtract_regions(base, case[1])
    local area = 0
    for _, r in ipairs(regions) do
      area = area + (r[3] - r[1]) * (r[4] - r[2])
    end
    assert(math.abs(area - case[2]) < 0.001, "region subtraction changed the allowed area")
    for x = 0.1, 9.9, 0.2 do
      for y = 0.1, 9.9, 0.2 do
        local expected = true
        for _, mask in ipairs(case[1]) do
          if x > mask[1] and x < mask[3] and y > mask[2] and y < mask[4] then expected = false end
        end
        assert(growth.in_regions(regions, x, y) == expected, "region union does not match its masks")
      end
    end
    local field = growth.field(regions)
    for i, x in ipairs(field.x) do
      assert(growth.in_regions(regions, x, field.y[i]), "growth field escaped its allowed regions")
    end
  end

  florilegium_test_case = 1
  growth.regions = function(g, layout)
    g.colright = g.pw - 0.3
    local row = { 0.3, 0.3, g.pw - 0.3, g.ph - 0.3 }
    local masks = {}
    if florilegium_test_case == 2 then
      local center = g.pw - (0.75 + 0.17 * (g.pw - 3.0))
      masks = { { center - 1.575, 1.3, center + 1.575, 4.45 } }
    elseif florilegium_test_case == 3 then
      local center = g.pw - (0.75 + 0.17 * (g.pw - 3.0))
      masks = {
        { 1.2, g.ph - 4.5, g.pw - 1.2, g.ph - 0.9 },
        { center - 1.575, 1.3, center + 1.575, 4.45 },
      }
    end
    local regions = growth.subtract_regions({ row }, masks)
    local foot = { row[1], row[2], row[3], row[2] + 1.0 }
    return regions, row, foot, nil, row[3], masks
  end
  florilegium.head = function() end
}
